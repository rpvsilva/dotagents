# The secret rules, one per line: `<name> <ERE>`. Read by lib.sh (the guard hooks, remote-agent*.sh)
# and by adapters/claude/telemetry/extract.py, so keep to ERE features bash `=~` and Python `re`
# share: groups, alternation, bracket classes, {n,m}. No \b, \d or [[:class:]].
#
# path: a file whose name looks like a secret (.env, keys, certificates, credentials, secrets/).
path (^|/)(\.env(\.[^/]*)?|[^/]+\.(env|pem|key|p12|pfx)|id_(rsa|dsa|ecdsa|ed25519)|\.netrc|\.aws/credentials)$|(^|/)secrets/
# example: a path that matches `path` but is only an example, so it passes.
example \.(example|sample|template)$
# token: text that looks like a token or private key; one alternative per line, joined with |.
token -----BEGIN ([A-Z]+ )?PRIVATE KEY-----
token gh[pousr]_[A-Za-z0-9]{36}
token github_pat_[A-Za-z0-9_]{22,}
token AKIA[0-9A-Z]{16}
token xox[abprs]-[A-Za-z0-9-]{10,}
token sk-ant-[A-Za-z0-9_-]{20,}
token AIza[0-9A-Za-z_-]{35}
# label: a secret in a step label, matched case-insensitively; only the telemetry extractor applies
# these, never the guard hooks. It masks each match of a token line as ***, and of a label line too,
# unless the label line has groups: then only its last group (the value, which must close last and
# always match) becomes ***, so the label keeps the program, flag, host and path. Applied in order.
# URL userinfo (https://user:pw@host) and query string (https://host/path?key=x):
label [a-z][a-z0-9+.-]*://([^ "'/?#@]+)@
label [a-z][a-z0-9+.-]*://[^ "'?]*[?]([^ "']+)
# curl -u / --user user:password:
label (^|[ "'])(-u|--user)( *|=)([^ "':]+:[^ "']+)
# mysql -psecret (a password glued to -p):
label (mysql[a-z]*|mariadb[a-z-]*)( [^ ]+)* -p([^ "']+)
# Authorization: <scheme> <value>, or a bare value:
label authorization *: *([a-z]+ +[^ "']+|[^ "']+)
label bearer [^ "']+
# --password value (a flag naming a secret, then its value after a space):
label --[a-z0-9_-]*(pass|pwd|secret|token|key)[a-z0-9_-]* +([^ "'-][^ "']*)
# NAME=value or NAME: value, where NAME names a secret (DB_PASS=, --password=, X-Api-Key:):
label [a-z0-9_-]*(pass|pwd|key|secret|token|auth)[a-z0-9_-]* *[=:] *([^ "']+)
# Token-shaped keys the token lines above leave out (OpenAI sk-..., Stripe sk_live_...):
label (^|[^a-z0-9])(sk-[a-z0-9_-]{20,}|[rs]k_(live|test)_[a-z0-9]{10,})
